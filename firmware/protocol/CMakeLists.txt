cmake_minimum_required(VERSION 3.16)
project(bsg_event_protocol C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(ENABLE_SANITIZERS "Host address/undefined sanitizers" OFF)
add_library(event_protocol event_protocol.c event_bridge.c)
add_library(nostos_protocol nostos_protocol.c nostos_uart.c nostos_bridge.c nostos_state.c nostos_endpoint.c nostos_debug.c)
target_include_directories(nostos_protocol PUBLIC .)
target_compile_options(nostos_protocol PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
target_include_directories(event_protocol PUBLIC .)
target_compile_options(event_protocol PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
  target_compile_options(nostos_protocol PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nostos_protocol PUBLIC -fsanitize=address,undefined)
  target_compile_options(event_protocol PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(event_protocol PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(test_event_protocol tests/test_event_protocol.c)
target_link_libraries(test_event_protocol PRIVATE event_protocol)
target_compile_options(test_event_protocol PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME event_protocol COMMAND test_event_protocol)
add_executable(test_event_bridge tests/test_event_bridge.c)
target_link_libraries(test_event_bridge PRIVATE event_protocol)
target_compile_options(test_event_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME event_bridge COMMAND test_event_bridge)
add_executable(test_nostos_safety tests/test_nostos_safety.c)
target_link_libraries(test_nostos_safety PRIVATE nostos_protocol)
target_compile_options(test_nostos_safety PRIVATE
  -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
add_test(NAME nostos_safety COMMAND test_nostos_safety)

add_library(sensor_link sensor_link.c)
target_include_directories(sensor_link PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(sensor_link PRIVATE
  -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
add_executable(test_sensor_link tests/test_sensor_link.c)
target_link_libraries(test_sensor_link PRIVATE sensor_link)
target_compile_options(test_sensor_link PRIVATE
  -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
add_test(NAME sensor_link COMMAND test_sensor_link)
