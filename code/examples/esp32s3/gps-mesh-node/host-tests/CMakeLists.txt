cmake_minimum_required(VERSION 3.16)
project(gps_mesh_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(test_gps test_gps.c ../main/gps_codec.c ../main/gps_receiver.c ../main/serial_command.c)
target_include_directories(test_gps PRIVATE ../main)
target_compile_options(test_gps PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
# Keep assert-based test checks active in Release as well as Debug.
target_compile_options(test_gps PRIVATE -UNDEBUG)
target_link_options(test_gps PRIVATE -fsanitize=address,undefined)
add_test(NAME gps_wire_receiver_and_serial COMMAND test_gps)
